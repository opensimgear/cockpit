cmake_minimum_required(VERSION 3.16)
set(OSG_BOARD_PROFILE "headless" CACHE STRING "Board profile: headless, waveshare-lcd7-s3 or sunton-3248s035r")
set_property(CACHE OSG_BOARD_PROFILE PROPERTY STRINGS headless waveshare-lcd7-s3 sunton-3248s035r)
if(OSG_BOARD_PROFILE STREQUAL "waveshare-lcd7-s3")
  if(DEFINED IDF_TARGET AND NOT IDF_TARGET STREQUAL "esp32s3")
    message(FATAL_ERROR "waveshare-lcd7-s3 requires IDF_TARGET=esp32s3")
  endif()
  set(IDF_TARGET "esp32s3" CACHE STRING "Target MCU")
  if(DEFINED OSG_BOARD_SOURCE AND NOT OSG_BOARD_SOURCE STREQUAL "${CMAKE_CURRENT_LIST_DIR}/boards/waveshare-lcd7/board.c")
    message(FATAL_ERROR "Board source conflicts with profile; use a fresh build directory")
  endif()
  set(OSG_BOARD_SOURCE "${CMAKE_CURRENT_LIST_DIR}/boards/waveshare-lcd7/board.c" CACHE FILEPATH "Board adapter")
  if(NOT DEFINED SDKCONFIG_DEFAULTS)
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
  endif()
  list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/boards/waveshare-lcd7/sdkconfig.defaults")
elseif(OSG_BOARD_PROFILE STREQUAL "sunton-3248s035r")
  if(DEFINED IDF_TARGET AND NOT IDF_TARGET STREQUAL "esp32")
    message(FATAL_ERROR "sunton-3248s035r requires IDF_TARGET=esp32")
  endif()
  set(IDF_TARGET "esp32" CACHE STRING "Target MCU")
  if(DEFINED OSG_BOARD_SOURCE AND NOT OSG_BOARD_SOURCE STREQUAL "${CMAKE_CURRENT_LIST_DIR}/boards/sunton-3248s035r/board.c")
    message(FATAL_ERROR "Board source conflicts with profile; use a fresh build directory")
  endif()
  set(OSG_BOARD_SOURCE "${CMAKE_CURRENT_LIST_DIR}/boards/sunton-3248s035r/board.c" CACHE FILEPATH "Board adapter")
  if(NOT DEFINED SDKCONFIG_DEFAULTS)
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
  endif()
  list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/boards/sunton-3248s035r/sdkconfig.defaults")
elseif(OSG_BOARD_PROFILE STREQUAL "headless")
  if(NOT DEFINED SDKCONFIG_DEFAULTS)
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
  endif()
  list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/boards/headless/sdkconfig.defaults")
else()
  message(FATAL_ERROR "Unknown OSG_BOARD_PROFILE")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(opensimgear_companion_esp32)
